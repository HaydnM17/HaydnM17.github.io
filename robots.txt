# Crawling is deliberately ALLOWED here. Every page carries
# <meta name="robots" content="noindex">, and a crawler has to be able to
# fetch a page to read that. Disallowing it would do the opposite of what is
# wanted: the pages could not be re-crawled, the noindex would never be seen,
# and anything already indexed would stay in the results as a bare link.
#
# The current site is https://haydnmcintyre.ca and that is the one to index.

Disallow: /api/
